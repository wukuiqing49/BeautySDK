cmake_minimum_required(VERSION 3.22.1)
project(beauty_gpupixel LANGUAGES CXX)
add_library(beauty_gpupixel SHARED beauty_bridge.cpp)
target_compile_features(beauty_gpupixel PRIVATE cxx_std_17)
target_compile_options(beauty_gpupixel PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(beauty_gpupixel PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(beauty_gpupixel PRIVATE GLESv3 EGL)
